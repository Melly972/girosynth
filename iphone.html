<body>

    <h1>GIRO</h1>

    <div id="status" class="status">Connexion au serveur...</div>

    <button id="startBtn" class="btn">ACTIVER LES CAPTEURS</button>


    <script>

        // Récupérer la session dans l'URL (?s=XXXX)

        const params = new URLSearchParams(window.location.search);

        const SESSION = params.get('s');

        const WS_URL = "wss://socketsbay.com/wss/v2/1/demo/";

        let ws = new WebSocket(WS_URL);


        ws.onopen = () => {

            document.getElementById('status').textContent = "Connecté ! Appuie sur le bouton.";

            // Envoyer un ping au Mac

            ws.send(JSON.stringify({ sid: SESSION, type: 'ping' }));

        };


        document.getElementById('startBtn').onclick = async () => {

            // ÉTAPE CRUCIALE POUR IOS : Demander la permission

            if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {

                try {

                    const permission = await DeviceOrientationEvent.requestPermission();

                    if (permission === 'granted') {

                        startTracking();

                    } else {

                        alert("Permission refusée");

                    }

                } catch (e) {

                    alert("Erreur permission : " + e);

                }

            } else {

                // Pour Android ou navigateurs sans système de permission

                startTracking();

            }

        };


        function startTracking() {

            document.getElementById('startBtn').style.display = 'none';

            document.getElementById('status').textContent = "Envoi des données au Mac...";

            

            window.addEventListener('deviceorientation', (e) => {

                const data = {

                    alpha: e.alpha || 0,

                    beta: e.beta || 0,

                    gamma: e.gamma || 0

                };

                

                if (ws.readyState === WebSocket.OPEN) {

                    ws.send(JSON.stringify({

                        sid: SESSION,

                        type: 'gyro',

                        data: data

                    }));

                }

            });

        }

    </script>
